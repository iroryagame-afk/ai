{MACD底背离 - 富途高亮版}

{一、MACD}
DIF:EMA(CLOSE,12)-EMA(CLOSE,26),COLORYELLOW,LINETHICK2;
DEA:EMA(DIF,9),COLORCYAN,LINETHICK2;
MACD值:=(DIF-DEA)*2;
零轴:0,COLORGRAY;

{二、四色动能柱}
STICKLINE(MACD值>=0 AND MACD值>=REF(MACD值,1),0,MACD值,2.5,0),COLORRED;
STICKLINE(MACD值>=0 AND MACD值<REF(MACD值,1),0,MACD值,2.5,0),COLORYELLOW;
STICKLINE(MACD值<0 AND MACD值<REF(MACD值,1),0,MACD值,2.5,0),COLORGREEN;
STICKLINE(MACD值<0 AND MACD值>=REF(MACD值,1),0,MACD值,2.5,0),COLORCYAN;

{三、底背离条件}
A1:=BARSLAST(REF(CROSS(DIF,DEA),1));
底背离:=A1>=5 AND A1<=60 AND DIF<0 AND REF(CLOSE,A1+1)>CLOSE AND DIF>REF(DIF,A1+1) AND CROSS(DIF,DEA);

{四、醒目标记}
标记幅度:=HHV(ABS(MACD值),60);
图标位置:=DIF-标记幅度*0.12;
文字位置:=DIF-标记幅度*0.30;

STICKLINE(底背离,0,DIF,4,0),COLORMAGENTA;
DRAWICON(底背离,图标位置,1);
DRAWTEXT(底背离,文字位置,'★★底背离★★'),COLORYELLOW;
