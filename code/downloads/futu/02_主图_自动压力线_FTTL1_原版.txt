SW:=3;
MAXEXT:=60;
TOL:=0.025;
VOLK:=1.20;

PH:=REF(H,SW)=HHV(H,SW*2+1) AND REF(H,SW)>REF(H,SW+1) AND REF(H,SW)>=REF(H,SW-1);
PL:=REF(L,SW)=LLV(L,SW*2+1) AND REF(L,SW)<REF(L,SW+1) AND REF(L,SW)<=REF(L,SW-1);

HASPH:=COUNT(PH,300)>=2;
HASPL:=COUNT(PL,300)>=2;

BH1:=BARSLAST(PH);
BL1:=BARSLAST(PL);
BH2:=BH1+REF(BARSLAST(PH),BH1+1)+1;
BL2:=BL1+REF(BARSLAST(PL),BL1+1)+1;

HA1:=BH1+SW;
HA2:=BH2+SW;
LA1:=BL1+SW;
LA2:=BL2+SW;

H1:=REF(H,HA1);
H2:=REF(H,HA2);
L1:=REF(L,LA1);
L2:=REF(L,LA2);

HDX:=HA2-HA1;
LDX:=LA2-LA1;
HSLOPE:=(H1-H2)/HDX;
LSLOPE:=(L1-L2)/LDX;

HLINE:=H1+HSLOPE*HA1;
LLINE:=L1+LSLOPE*LA1;

HAVG:=(H1+H2)/2;
LAVG:=(L1+L2)/2;

DRES:=HASPH AND H1<H2 AND HA1<=MAXEXT;
ASUP:=HASPL AND L1>L2 AND LA1<=MAXEXT;
FLATRES:=HASPH AND ABS(H1-H2)/HAVG<=TOL AND HA1<=MAXEXT;
FLATSUP:=HASPL AND ABS(L1-L2)/LAVG<=TOL AND LA1<=MAXEXT;

DOWN_RES:IF(DRES,HLINE,DRAWNULL),LINETHICK2,COLORRED;
UP_SUP:IF(ASUP,LLINE,DRAWNULL),LINETHICK2,COLOR00C853;
FLAT_RES:IF(FLATRES,HAVG,DRAWNULL),DOTLINE,COLORFF8C00;
FLAT_SUP:IF(FLATSUP,LAVG,DRAWNULL),DOTLINE,COLOR00BFFF;

SYMTRI:=DRES AND ASUP AND HLINE>LLINE;
ASCTRI:=FLATRES AND ASUP AND NOT(SYMTRI);
DESCTRI:=DRES AND FLATSUP AND NOT(SYMTRI);
BOXPAT:=FLATRES AND FLATSUP AND NOT(ASCTRI) AND NOT(DESCTRI);

DRAWTEXT(SYMTRI AND NOT(REF(SYMTRI,1)),H*1.01,'TRI?'),COLORMAGENTA;
DRAWTEXT(ASCTRI AND NOT(REF(ASCTRI,1)),H*1.01,'ASC-TRI?'),COLORRED;
DRAWTEXT(DESCTRI AND NOT(REF(DESCTRI,1)),H*1.01,'DESC-TRI?'),COLORFF8C00;
DRAWTEXT(BOXPAT AND NOT(REF(BOXPAT,1)),H*1.01,'BOX?'),COLORYELLOW;

BRDRES:=DRES AND C>HLINE AND REF(C,1)<=REF(HLINE,1);
BRFRES:=FLATRES AND C>HAVG AND REF(C,1)<=REF(HAVG,1);
BRASUP:=ASUP AND C<LLINE AND REF(C,1)>=REF(LLINE,1);
BRFSUP:=FLATSUP AND C<LAVG AND REF(C,1)>=REF(LAVG,1);

VOLUP:=V>MA(V,20)*VOLK;

DRAWTEXT((BRDRES OR BRFRES) AND VOLUP,L*0.995,'BREAK-UP'),COLORMAGENTA;
DRAWTEXT((BRDRES OR BRFRES) AND NOT(VOLUP),L*0.995,'BREAK?'),COLORYELLOW;
DRAWTEXT(BRASUP OR BRFSUP,H*1.005,'BREAK-DN'),COLORRED;
