{BIAS20/60 五档拥挤度·冷暖视觉版}
{必须使用日K}
{真实信号不变，仅优化副图显示}

MA20:=MA(CLOSE,20);
MA60:=MA(CLOSE,60);

BIAS20_RAW:=(CLOSE-MA20)/MA20*100;
BIAS60_RAW:=(CLOSE-MA60)/MA60*100;

{限制极端值对纵轴的挤压}
BIAS20_VIEW:=MIN(25,MAX(-10,BIAS20_RAW));
BIAS60_VIEW:=MIN(25,MAX(-10,BIAS60_RAW));

BIAS20:ROUND2(BIAS20_VIEW,1),COLOR00AEEF,LINETHICK2;
BIAS60:ROUND2(BIAS60_VIEW,1),COLORFF6B6B,LINETHICK2;

{五档冷暖线}
EXTREME_COLD:-8,COLORBLUE,DOTLINE;
COLD:-5,COLORCYAN,DOTLINE;
WARN:10,COLORYELLOW,DOTLINE;
HOT:15,COLORRED,DOTLINE;
EXTREME:20,COLORMAGENTA,DOTLINE;

{标签错位，避免堆叠}
DRAWTEXT(CURRBARSCOUNT=1,-9,'极冷-8'),COLORBLUE;
DRAWTEXT(CURRBARSCOUNT=1,-4,'冷-5'),COLORCYAN;
DRAWTEXT(CURRBARSCOUNT=1,8,'警戒10'),COLORYELLOW;
DRAWTEXT(CURRBARSCOUNT=1,15,'过热15'),COLORRED;
DRAWTEXT(CURRBARSCOUNT=1,22,'极热20'),COLORMAGENTA;
